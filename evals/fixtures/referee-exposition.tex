\title{A finite polynomial identity}
\begin{document}
\begin{abstract}
We give a polynomial formula for the sum of the first $n$ positive cubes.
\end{abstract}
\section{The identity}
\begin{theorem}\label{thm:cubes}
For every integer $n\geq0$,
$\sum_{j=1}^n j^3=n^2(n+1)^2/4$.
\end{theorem}
\begin{proof}
Put $A=\sum_{j=1}^n j$, $B=\sum_{j=1}^n j^2$, $C=\sum_{j=1}^n j^3$.
$j^2-(j-1)^2=2j-1$.
$n^2=2A-n$.
$A=(n^2+n)/2$.
$j^3-(j-1)^3=3j^2-3j+1$.
$n^3=3B-3A+n$.
$B=n^3/3+(n^2+n)/2-n/3$.
$B=(2n^3+3n^2+n)/6$.
$j^4-(j-1)^4=4j^3-6j^2+4j-1$.
$n^4=4C-6B+4A-n$.
$4C=n^4+(2n^3+3n^2+n)-2(n^2+n)+n$.
$4C=n^4+2n^3+n^2$.
$C=n^2(n+1)^2/4$.
\end{proof}
