\title{A coefficient extension}
\begin{document}
\begin{abstract}
An invariant of finite complexes vanishes with arbitrary coefficient rings.
\end{abstract}
\section{Conventions}
For a finite complex $X$ and a commutative coefficient ring $R$, write
$V(X;R)$ for the invariant defined in the cited construction.
The cited source is not supplied with this manuscript.
\section{Extension}
\begin{theorem}\label{thm:extension}
If $X$ satisfies condition P, then $V(X;R)=0$ for every commutative ring $R$.
\end{theorem}
\begin{proof}
Theorem 3 of \cite{coefficient-note} gives the conclusion for rational
coefficients. The same theorem therefore proves the assertion for arbitrary
coefficient rings.
\end{proof}
